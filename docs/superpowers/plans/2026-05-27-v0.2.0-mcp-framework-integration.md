```python
"""Emerald MCP Server — exposes memory operations as MCP tools."""

from __future__ import annotations

import os

from fastmcp import FastMCP
from emerald.sdk.client import EmeraldClient

mcp = FastMCP("emerald")

_client: EmeraldClient | None = None


def _get_client() -> EmeraldClient:
    global _client
    if _client is None:
        api_key = os.environ["EMERALD_API_KEY"]
        base_url = os.environ.get("EMERALD_BASE_URL", "http://localhost:8000")
        _client = EmeraldClient(api_key=api_key, base_url=base_url)
    return _client


@mcp.tool()
async def emerald_add(content: str, entity_id: str) -> dict:
    """Add a memory to Emerald.

    Args:
        content: The text content to remember.
        entity_id: The entity (user/project) to associate the memory with.
    """
    client = _get_client()
    result = await client.add(content=content, entity_id=entity_id)
    return {"memory_ids": result.memory_ids, "status": result.pipeline_status}


@mcp.tool()
async def emerald_search(q: str, entity_id: str, search_mode: str = "hybrid") -> dict:
    """Search memories and documents in Emerald.

    Args:
        q: The search query.
        entity_id: The entity to search within.
        search_mode: One of "hybrid" (default), "memory", or "rag".
    """
    client = _get_client()
    result = await client.search(q=q, entity_id=entity_id, search_mode=search_mode)
    return {
        "results": [
            {"id": r.id, "content": r.content, "score": r.score, "source": r.source}
            for r in result.results
        ],
        "search_mode": result.search_mode,
    }


@mcp.tool()
async def emerald_profile(entity_id: str) -> dict:
    """Get the user profile from Emerald.

    Returns static facts (long-term) and dynamic facts (recent context).
    """
    client = _get_client()
    result = await client.profile(entity_id=entity_id)
    return {
        "entity_id": result.entity_id,
        "static": [f.content for f in result.static],
        "dynamic": [f.content for f in result.dynamic],
        "memory_count": result.memory_count,
    }


if __name__ == "__main__":
    mcp.run(transport="stdio")
```

```typescript
// packages/sdk-ts/src/client.ts
export interface EmeraldClientOptions {
  apiKey: string;
  baseUrl?: string;
}

export class EmeraldClient {
  private apiKey: string;
  private baseUrl: string;

  constructor(options: EmeraldClientOptions) {
    this.apiKey = options.apiKey;
    this.baseUrl = options.baseUrl || "http://localhost:8000";
  }

  private async request(path: string, body?: unknown): Promise<unknown> {
    const response = await fetch(`${this.baseUrl}${path}`, {
      method: body ? "POST" : "GET",
      headers: {
        "Content-Type": "application/json",
        Authorization: `Bearer ${this.apiKey}`,
      },
      ...(body ? { body: JSON.stringify(body) } : {}),
    });
    if (!response.ok) {
      throw new Error(`Emerald API error: ${response.status} ${response.statusText}`);
    }
    return response.json();
  }

  async add(params: { content: string; entityId: string; contentType?: string }): Promise<unknown> {
    return this.request("/v1/memories", {
      content: params.content,
      entity_id: params.entityId,
      content_type: params.contentType || "text",
    });
  }

  async search(params: { q: string; entityId: string; searchMode?: string; topK?: number }): Promise<unknown> {
    return this.request("/v1/search", {
      q: params.q,
      entity_id: params.entityId,
      search_mode: params.searchMode || "hybrid",
      top_k: params.topK || 10,
    });
  }

  async profile(params: { entityId: string }): Promise<unknown> {
    return this.request(`/v1/profiles/${params.entityId}`);
  }
}
```

```typescript
// packages/sdk-ts/src/vercel.ts
import { EmeraldClient } from "./client";

export interface WithEmeraldOptions {
  apiKey: string;
  baseUrl?: string;
  entityId: string;
  mode?: "profile" | "query" | "full";
}

export function withEmerald(options: WithEmeraldOptions) {
  const client = new EmeraldClient({
    apiKey: options.apiKey,
    baseUrl: options.baseUrl,
  });

  return {
    wrapGenerate: async ({ doGenerate, params }: any) => {
      const queryText = extractLastUserMessage(params.messages);

      const response = (await client.search({
        q: queryText || "",
        entityId: options.entityId,
        searchMode: "hybrid",
      })) as any;

      const memories = formatMemoriesForPrompt(response);
      const augmentedMessages = injectMemories(params.messages, memories);

      return doGenerate({ ...params, messages: augmentedMessages });
    },
  };
}

function extractLastUserMessage(messages: any[]): string {
  const userMsg = [...messages].reverse().find((m) => m.role === "user");
  if (!userMsg) return "";
  return typeof userMsg.content === "string"
    ? userMsg.content
    : userMsg.content
        ?.filter((p: any) => p.type === "text")
        .map((p: any) => p.text)
        .join(" ") || "";
}

function formatMemoriesForPrompt(response: any): string {
  const results = response.data?.results || [];
  if (results.length === 0) return "";
  return "Relevant memories:\n" + results.map((r: any) => `- ${r.content}`).join("\n");
}

function injectMemories(messages: any[], memories: string): any[] {
  if (!memories) return messages;
  const systemMsg = messages.find((m) => m.role === "system");
  if (systemMsg) {
    systemMsg.content = `${systemMsg.content}\n\n${memories}`;
    return messages;
  }
  return [{ role: "system", content: memories }, ...messages];
}
```

```python
"""LangChain integration for Emerald."""

from __future__ import annotations

from langchain.tools import BaseTool

from emerald.sdk.client import EmeraldClient


class EmeraldSearchTool(BaseTool):
    name = "emerald_search"
    description = "Search the user's memories and documents. Use this when you need to recall something the user previously told you or find relevant documents."

    def __init__(self, client: EmeraldClient, entity_id: str):
        super().__init__()
        self._client = client
        self._entity_id = entity_id

    def _run(self, query: str) -> str:
        result = self._client.search(q=query, entity_id=self._entity_id, search_mode="hybrid")
        if not result.results:
            return "No relevant memories found."
        lines = [f"- {r.content} (score: {r.score:.2f})" for r in result.results[:5]]
        return "Relevant memories:\n" + "\n".join(lines)

    async def _arun(self, query: str) -> str:
        return self._run(query)


class EmeraldAddTool(BaseTool):
    name = "emerald_add"
    description = "Add a memory for the user. Use this when the user shares important information you should remember."

    def __init__(self, client: EmeraldClient, entity_id: str):
        super().__init__()
        self._client = client
        self._entity_id = entity_id

    def _run(self, content: str) -> str:
        result = self._client.add(content=content, entity_id=self._entity_id)
        return f"Memory added: {result.memory_ids[0]}"

    async def _arun(self, content: str) -> str:
        return self._run(content)
```
